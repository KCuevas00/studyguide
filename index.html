    Sample Program to Write
    
          
    
            Read a file using for x in file and a context manager (with...) that:
            
    1. Prints the number of lines in the file.
            
    2. Prints the number of empty lines (lines containing only \n).
            
    3. Prints the number of characters in the file (not counting \n).
            
    4. Uses a regex to count or print the lines that contain a specific pattern.
          
    

          
    
            
            
    import re

pattern = re.compile(r'python', re.IGNORECASE)

total_lines = 0
empty_lines = 0
total_chars = 0
matching_lines = 0

filename = 'files-many.txt'

with open(filename, 'r', encoding='utf-8') as f:
    for line in f:
        total_lines += 1
        
        # 1. Check if empty line (only newline)
        if line == '\n' or line == '\r\n':
            empty_lines += 1
            
        # 2. Count characters without end-of-line markers
        stripped = line.rstrip('\r\n')
        total_chars += len(stripped)
        
        # 3. Regex match on line
        if pattern.search(line):
            matching_lines += 1

print(f"Total lines: {total_lines}")
print(f"Empty lines: {empty_lines}")
print(f"Total characters (not counting \\n): {total_chars}")
print(f"Matching lines: {matching_lines}")
